# Pixel-Männchen (PixelGuy) Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Ein SVG Pixel-Art Männchen das beim Seitenlade und bei bestimmten Aktionen mit einer Sprechblase erscheint, einen witzigen oder liebevollen Spruch zeigt, und nach 10 Sekunden automatisch verschwindet.

**Architecture:** Selbstständiger `<script>`-Block am Ende von `index.html` — kein Framework, kein externer Code. Das Männchen ist ein inline SVG (32×32 Grid, 4× skaliert). Ein `<div id="pixelguy-overlay">` wird dynamisch in `<body>` eingefügt. Trigger werden durch Wrapping bestehender Spar-Funktionen eingehängt (kein Eingriff in die Logik selbst).

**Tech Stack:** Vanilla JS, inline SVG, CSS transitions, bestehende Tailwind-CDN-Umgebung

function buildSVG() {
  const W = 32, H = 32, S = PIXEL_SIZE;
  let rects = '';
  for (const [x, y, fill] of PIXELS) {
    rects += `<rect x="${x*S}" y="${y*S}" width="${S}" height="${S}" fill="${fill}"/>`;
  }
  return `<svg xmlns="http://www.w3.org/2000/svg" width="${W*S}" height="${H*S}" shape-rendering="crispEdges" style="image-rendering:pixelated">${rects}</svg>`;
}

function randomSpruch(context) {
  const pool = SPRUECHE[context] || SPRUECHE.pageload;
  return pool[Math.floor(Math.random() * pool.length)];

// ── Overlay DOM ───────────────────────────────────────────────────────────
let overlayEl = null;
let hideTimer = null;

function ensureOverlay() {
  if (overlayEl) return;
  overlayEl = document.createElement('div');
  overlayEl.id = 'pixelguy-overlay';
  overlayEl.style.cssText = [
    'position:fixed',
    'z-index:9999',
    'pointer-events:none',
    'display:none',
    'flex-direction:column',
    'align-items:center',
    'gap:6px',
    'transition:opacity 0.4s ease',
    'opacity:0',
  ].join(';');
  document.body.appendChild(overlayEl);
}

function randomPosition(overlayW, overlayH) {
  const margin = 20;
  const maxX = window.innerWidth  - overlayW - margin;
  const maxY = window.innerHeight - overlayH - margin;
  const x = margin + Math.floor(Math.random() * Math.max(1, maxX - margin));
  const y = margin + Math.floor(Math.random() * Math.max(1, maxY - margin));
  return {x, y};
}

// ── Öffentliche API ───────────────────────────────────────────────────────
const pixelguy = {
  show(context) {
    ensureOverlay();
    if (hideTimer) { clearTimeout(hideTimer); hideTimer = null; }

    const text = randomSpruch(context);
    overlayEl.innerHTML = buildBubble(text) + buildSVG();
    overlayEl.style.display = 'flex';

    // Position nach nächstem Frame berechnen (damit offsetWidth stimmt)
    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        const {x, y} = randomPosition(overlayEl.offsetWidth || 220, overlayEl.offsetHeight || 180);
        overlayEl.style.left = x + 'px';
        overlayEl.style.top  = y + 'px';
        overlayEl.style.opacity = '1';
      });
    });

    hideTimer = setTimeout(() => pixelguy.hide(), 10000);
  },

  maybeShow(context, probability = 0.4) {
    if (Math.random() < probability) this.show(context);
  },

  hide() {
    if (!overlayEl) return;
    overlayEl.style.opacity = '0';
    setTimeout(() => { if (overlayEl) overlayEl.style.display = 'none'; }, 450);
    if (hideTimer) { clearTimeout(hideTimer); hideTimer = null; }
  },
};

// ── Trigger: Seitenlade ───────────────────────────────────────────────────

  pixelguy.show('pageload');
});

// ── Trigger: Note vergeben ────────────────────────────────────────────────
// Wir wrappen saveDataWithUndo so dass nach dem Speichern geprüft werden kann,
// welche Note vergeben wurde. Die Note wird via pixelguy._pendingNote gesetzt.
const _origSaveDataWithUndo = window.saveDataWithUndo;
if (typeof _origSaveDataWithUndo === 'function') {
  window.saveDataWithUndo = function() {
    _origSaveDataWithUndo.apply(this, arguments);
    const n = pixelguy._pendingNote;
    pixelguy._pendingNote = null;
    if (n === 1) pixelguy.maybeShow('note_1');
    else if (n === 5) pixelguy.maybeShow('note_5');
  };
}

// Hilfsfunktion: wird von den Render-Input-Handlern aufgerufen
window.pixelguyNoteHint = function(note) {
  pixelguy._pendingNote = +note;
};

window.pixelguy = pixelguy;
})();
